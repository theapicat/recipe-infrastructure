1. Designdisiplin & 60-30-10 Regelen

60% Base

Krem og havre i Light mode, mørk skogsbunn i Dark mode. Gir en behagelig og varm bakgrunn for matbilder.

30% Primary (Sage)

Dempede mose- og salviegrønne toner. Brukes til navigasjon, primærknapper og visuell struktur.

10% Accent (Terracotta)

Varm terrakotta og brent jord. Brukes sparsommelig på blikkfang, kategorier og fremhevede tilstander.

5. TypeScript Konfigurasjon (theme.ts)

import { createTheme, MantineColorsTuple } from '@mantine/core';

const sage: MantineColorsTuple = [
  '#f2f6f3', '#e3ece5', '#c5d9c9', '#a4c4ab', '#7ca386',
  '#5e8869', '#4a6b53', '#3a5441', '#2a3e30', '#18261c'
];

const terracotta: MantineColorsTuple = [
  '#fdf5f3', '#f9ebe6', '#f1d3ca', '#e6b2a2', '#dd8a6e',
  '#d07052', '#c86a4b', '#a34e34', '#7e3a25', '#4e2114'
];

export const theme = createTheme({
  primaryColor: 'sage',
  primaryShade: { light: 6, dark: 4 },
  colors: { sage, terracotta },
  defaultRadius: 'md',
  fontFamily: '-apple-system, BlinkMacSystemFont, "Segoe UI", Roboto, sans-serif',
});
